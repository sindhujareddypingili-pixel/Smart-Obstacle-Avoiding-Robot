void setMotorSpeed(uint8_t l, uint8_t r){
  ledcWrite(CH_A, l);
  ledcWrite(CH_B, r);
}

void driveStop(){
  digitalWrite(IN1, LOW);  digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);  digitalWrite(IN4, LOW);
  setMotorSpeed(0, 0);
}

// Motor A forward, Motor B forward → car moves forward
void driveForward(uint8_t s){
  digitalWrite(IN1, HIGH); digitalWrite(IN2, LOW);
  digitalWrite(IN3, HIGH); digitalWrite(IN4, LOW);
  setMotorSpeed(s, s);
}

// Motor A backward, Motor B backward → car moves backward
void driveBackward(uint8_t s){
  digitalWrite(IN1, LOW);  digitalWrite(IN2, HIGH);
  digitalWrite(IN3, LOW);  digitalWrite(IN4, HIGH);
  setMotorSpeed(s, s);
}

// Motor A backward, Motor B forward → car turns left
void driveLeft(uint8_t s){
  digitalWrite(IN1, LOW);  digitalWrite(IN2, HIGH);
  digitalWrite(IN3, HIGH); digitalWrite(IN4, LOW);
  setMotorSpeed(s, s);
}

// Motor A forward, Motor B backward → car turns right
void driveRight(uint8_t s){
  digitalWrite(IN1, HIGH); digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW);  digitalWrite(IN4, HIGH);
  setMotorSpeed(s, s);
}

// ================= Ultrasonic =================
float getDistanceCm(){
  digitalWrite(TRIG_PIN, LOW);  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  unsigned long d = pulseIn(ECHO_PIN, HIGH, 30000);
  if(d == 0) return -1;
  return d * 0.0343 / 2.0;
}

// ================= Manual Evade (non-blocking) =================
// Called every loop() when manual mode is active and evade guard is ON.
// State machine: NONE → BACKWARD → TURN → DONE → NONE
void handleManualEvade(){
  unsigned long now = millis();

  switch(mevadeState){

    case MEVADE_NONE:
      // Check if forward is pressed and obstacle is close
      if(forwardCmd && manualObstacleGuard){
        float d = getDistanceCm();
        if(d > 0 && d < OBSTACLE_DIST_CM){
          // Obstacle detected! Start evade sequence
          forwardCmd = false;          // cancel forward command
          driveBackward(autoSpeed);
          mevadeState  = MEVADE_BACKWARD;
          mevadeStart  = now;
          // Random turn direction using millis as seed
          mevadeTurnLeft = ((millis() % 2) == 0);
          Serial.println("EVADE: obstacle detected, reversing...");
        }
      }
      break;

    case MEVADE_BACKWARD:
      driveBackward(autoSpeed);
      if(now - mevadeStart >= MEVADE_BACK_TIME){
        // Switch to turn
        mevadeState = MEVADE_TURN;
        mevadeStart = now;
        Serial.print("EVADE: turning ");
        Serial.println(mevadeTurnLeft ? "LEFT" : "RIGHT");
      }
      break;

    case MEVADE_TURN:
      if(mevadeTurnLeft) driveLeft(autoSpeed);
      else               driveRight(autoSpeed);
      if(now - mevadeStart >= MEVADE_TURN_TIME){
        // Done — stop and hand back control
        driveStop();
        mevadeState = MEVADE_DONE;
        Serial.println("EVADE: done, control returned to manual");
      }
      break;

    case MEVADE_DONE:
      // Stay stopped, reset state so next loop is clean
      mevadeState = MEVADE_NONE;
      break;
  }
}

// ================= Manual (normal) =================
void handleManual(){
  // If evade is in progress, let handleManualEvade run instead
  if(mevadeState != MEVADE_NONE){
    handleManualEvade();
    return;
  }

  // Normal manual driving
  if(forwardCmd){
    // Pre-check obstacle guard before moving forward
    if(manualObstacleGuard){
      float d = getDistanceCm();
      if(d > 0 && d < OBSTACLE_DIST_CM){
        // Trigger evade immediately
        forwardCmd  = false;
        driveBackward(autoSpeed);
        mevadeState = MEVADE_BACKWARD;
        mevadeStart = millis();
        mevadeTurnLeft = ((millis() % 2) == 0);
        Serial.println("EVADE: triggered from handleManual");
        return;
      }
    }
    driveForward(manualSpeed);
  }
  else if(backCmd)  driveBackward(manualSpeed);
  else if(leftCmd)  driveLeft(manualSpeed);
  else if(rightCmd) driveRight(manualSpeed);
  else              driveStop();
}

// ================= Auto Obstacle =================
void handleObstacle(){
  unsigned long now = millis();
  switch(autoState){
    case AUTO_IDLE:
      autoState = AUTO_FORWARD;
      break;
    case AUTO_FORWARD:{
      float d = getDistanceCm();
      if(d > 0 && d < OBSTACLE_DIST_CM){
        driveStop();
        autoState     = AUTO_BACKWARD;
        autoStateStart = now;
      } else driveForward(autoSpeed);
    }break;
    case AUTO_BACKWARD:
      if(now - autoStateStart >= BACK_TIME){
        autoState     = AUTO_TURN;
        autoStateStart = now;
      } else driveBackward(autoSpeed);
      break;
    case AUTO_TURN:
      if(turnLeftNext) driveLeft(autoSpeed); else driveRight(autoSpeed);
      if(now - autoStateStart >= TURN_TIME){
        driveStop();
        turnLeftNext = !turnLeftNext;
        autoState    = AUTO_FORWARD;
      }
      break;
  }
}

// ================= Follow =================
void handleFollow(){
  float d = getDistanceCm();
  if(d < 0)        { driveStop(); return; }
  if(d < 12)         driveBackward(autoSpeed);
  else if(d <= 20)   driveStop();
  else if(d <= 45)   driveForward(autoSpeed);
  else               driveStop();
}

// ================= HTTP Handlers =================
void handleRoot()    { server.send_P(200, "text/html", index_html); }

void handleCmd(){
  // Block commands during obstacle mode, follow mode, or manual evade
  if(obstacleMode || followMode || mevadeState != MEVADE_NONE){
    server.send(200, "text/plain", "IGNORED");
    return;
  }
  String d = server.arg("dir");
  bool   p = server.arg("state").toInt();
  if     (d=="f") forwardCmd = p;
  else if(d=="b") backCmd    = p;
  else if(d=="l") leftCmd    = p;
  else if(d=="r") rightCmd   = p;
  else if(d=="s"){ forwardCmd=backCmd=leftCmd=rightCmd=false; driveStop(); }
  server.send(200, "text/plain", "OK");
}

void handleMode(){
  obstacleMode  = server.arg("auto").toInt();
  followMode    = false;
  autoState     = AUTO_FORWARD;
  mevadeState   = MEVADE_NONE;
  forwardCmd=backCmd=leftCmd=rightCmd=false;
  driveStop();
  server.send(200, "text/plain", "OK");
}

void handleFollowHttp(){
  followMode   = server.arg("f").toInt();
  obstacleMode = false;
  mevadeState  = MEVADE_NONE;
  forwardCmd=backCmd=leftCmd=rightCmd=false;
  driveStop();
  server.send(200, "text/plain", "OK");
}

void handleSpeed(){
  int v = server.arg("v").toInt();
  if(v >= 100 && v <= 255) manualSpeed = (uint8_t)v;
  server.send(200, "text/plain", "OK");
}

void handleManualObs(){
  manualObstacleGuard = server.arg("v").toInt();
  // Reset evade state when toggled off
  if(!manualObstacleGuard) mevadeState = MEVADE_NONE;
  server.send(200, "text/plain", "OK");
}

void handleDist(){
  float d = getDistanceCm();
  server.send(200, "text/plain", String(d, 1));
}

// Returns current manual evade state (0 = idle, 1 = evading)
// The UI polls this every 300ms to show/hide the evade banner
void handleEvadeState(){
  int s = (mevadeState != MEVADE_NONE && mevadeState != MEVADE_DONE) ? 1 : 0;
  server.send(200, "text/plain", String(s));
}

// ================= SETUP =================
void setup(){
  Serial.begin(115200);

  pinMode(IN1, OUTPUT); pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT); pinMode(IN4, OUTPUT);

  ledcSetup(CH_A, PWM_FREQ, PWM_RES);
  ledcAttachPin(ENA_PIN, CH_A);
  ledcSetup(CH_B, PWM_FREQ, PWM_RES);
  ledcAttachPin(ENB_PIN, CH_B);

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);

  // Seed random using floating ADC pin for better randomness
  randomSeed(analogRead(34));

  WiFi.mode(WIFI_AP);
  WiFi.softAP(apSsid, apPassword);
  Serial.print("AP IP: ");
  Serial.println(WiFi.softAPIP());

  server.on("/",           handleRoot);
  server.on("/cmd",        handleCmd);
  server.on("/mode",       handleMode);
  server.on("/follow",     handleFollowHttp);
  server.on("/speed",      handleSpeed);
  server.on("/manualobs",  handleManualObs);
  server.on("/dist",       handleDist);
  server.on("/evadestate", handleEvadeState);
  server.begin();
  Serial.println("Server started");
}

// ================= LOOP =================
void loop(){
  server.handleClient();

  if(followMode)        handleFollow();
  else if(obstacleMode) handleObstacle();
  else                  handleManual();   // includes evade logic

  delay(5);
}
